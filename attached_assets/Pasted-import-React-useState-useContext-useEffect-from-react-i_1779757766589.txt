import React,{
useState,
useContext,
useEffect
} from "react";

import {
View,
Text,
TouchableOpacity,
StyleSheet,
TextInput
} from "react-native";

import {AppContext} from "../context/AppContext";

export default function FocusScreen(){

const {
sessionLength,
timeSaved,
goal,
update
}=useContext(AppContext);

const [seconds,setSeconds]=useState(0);
const [locked,setLocked]=useState(false);

const [challengeType,setChallengeType]=useState("");
const [answer,setAnswer]=useState("");
const [solved,setSolved]=useState(false);

const puzzles=[
{
q:"8 + 7 = ?",
a:"15"
},
{
q:"12 - 5 = ?",
a:"7"
},
{
q:"3 x 4 = ?",
a:"12"
}
];

const reflections=[
"Why are you still scrolling?",
"What were you supposed to do next?",
"Will 10 more mins help your goal?"
];

useEffect(()=>{

const timer=setInterval(()=>{

setSeconds(prev=>{

const next=prev+1;

if(next>=sessionLength*60&&!locked){

setLocked(true);

const rand=Math.random();

if(rand<0.5){
setChallengeType("puzzle");
}else{
setChallengeType("reflection");
}

update({
goal,
strictness:"Balanced",
sessionLength,
timeSaved:timeSaved+sessionLength,
onboarded:true
});
}

return next;

});

},1000);

return()=>clearInterval(timer);

},[]);

const puzzle=
puzzles[Math.floor(Math.random()*puzzles.length)];

const reflection=
reflections[Math.floor(Math.random()*reflections.length)];

const verify=()=>{

if(
challengeType==="puzzle" &&
answer===puzzle.a
){
setSolved(true);
}

if(
challengeType==="reflection" &&
answer.length>10
){
setSolved(true);
}
};

const reset=()=>{
setLocked(false);
setSolved(false);
setSeconds(0);
setAnswer("");
};

const options=[5,15,30,45,60,90,120];

return(
<View style={styles.container}>

<Text style={styles.title}>
Session: {sessionLength} mins
</Text>

<View style={styles.row}>
{options.map(opt=>(
<TouchableOpacity
key={opt}
style={styles.option}
onPress={()=>update({
goal,
strictness:"Balanced",
sessionLength:opt,
timeSaved,
onboarded:true
})}
>
<Text style={styles.text}>{opt}</Text>
</TouchableOpacity>
))}
</View>

{locked&&(
<View style={styles.lockBox}>

<Text style={styles.lock}>
INTERVENTION
</Text>

<Text style={styles.prompt}>
{challengeType==="puzzle"
? puzzle.q
: reflection}
</Text>

<TextInput
style={styles.input}
placeholder="Answer..."
placeholderTextColor="#666"
value={answer}
onChangeText={setAnswer}
/>

<TouchableOpacity
style={styles.button}
onPress={verify}
>
<Text style={styles.text}>
Submit
</Text>
</TouchableOpacity>

{solved&&(
<TouchableOpacity
style={styles.unlock}
onPress={reset}
>
<Text style={styles.text}>
Resume Intentionally
</Text>
</TouchableOpacity>
)}

</View>
)}

</View>
);
}

const styles=StyleSheet.create({
container:{
flex:1,
backgroundColor:"#050505",
justifyContent:"center",
alignItems:"center"
},
title:{
color:"#4DA6FF",
fontSize:24
},
row:{
flexDirection:"row",
flexWrap:"wrap",
justifyContent:"center",
marginTop:30
},
option:{
backgroundColor:"#111",
padding:14,
margin:8,
borderRadius:12
},
lockBox:{
marginTop:40,
width:"85%",
alignItems:"center"
},
lock:{
fontSize:28,
color:"#4DA6FF",
fontWeight:"700"
},
prompt:{
color:"#fff",
marginVertical:20,
fontSize:18,
textAlign:"center"
},
input:{
backgroundColor:"#111",
color:"#fff",
width:"100%",
padding:16,
borderRadius:12
},
button:{
marginTop:20,
backgroundColor:"#4DA6FF",
padding:16,
borderRadius:12
},
unlock:{
marginTop:20,
backgroundColor:"#222",
padding:16,
borderRadius:12
},
text:{
color:"#fff"
}
});
