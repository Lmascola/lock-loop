import React,{
useState,
useContext,
useEffect
} from "react";

import {
View,
Text,
TouchableOpacity,
StyleSheet,
TextInput
} from "react-native";

import {AppContext} from "../context/AppContext";

export default function FocusScreen(){

const {
sessionLength,
timeSaved,
goal,
strictness,
update
}=useContext(AppContext);

const [seconds,setSeconds]=useState(0);
const [locked,setLocked]=useState(false);
const [answer,setAnswer]=useState("");
const [solved,setSolved]=useState(false);

const goalPrompts={

Focus:[
"What task were you meant to finish?",
"What is your next productive action?"
],

"Sleep Better":[
"What time are you sleeping tonight?",
"Will scrolling help tomorrow?"
],

Study:[
"What topic were you revising?",
"Write one thing you learned."
],

"Reduce Addiction":[
"Why are you opening this again?",
"What feeling are you avoiding?"
]

};

const puzzles=[
{q:"14 - 6 = ?",a:"8"},
{q:"7 x 3 = ?",a:"21"},
{q:"25 / 5 = ?",a:"5"}
];

const selectedPrompt=
goalPrompts[goal][
Math.floor(
Math.random()*goalPrompts[goal].length
)
];

const selectedPuzzle=
puzzles[
Math.floor(Math.random()*puzzles.length)
];

const usePuzzle=
strictness==="Aggressive";

useEffect(()=>{

const timer=setInterval(()=>{

setSeconds(prev=>{

const next=prev+1;

if(next>=sessionLength*60&&!locked){

setLocked(true);

update({
goal,
strictness,
sessionLength,
timeSaved:timeSaved+sessionLength,
onboarded:true
});

}

return next;

});

},1000);

return()=>clearInterval(timer);

},[]);

const verify=()=>{

if(usePuzzle){
if(answer===selectedPuzzle.a){
setSolved(true);
}
}else{
if(answer.length>10){
setSolved(true);
}
}

};

const reset=()=>{
setLocked(false);
setSolved(false);
setSeconds(0);
setAnswer("");
};

const options=[5,15,30,45,60,90,120];

return(
<View style={styles.container}>

<Text style={styles.title}>
{goal} Session: {sessionLength}m
</Text>

<View style={styles.row}>
{options.map(opt=>(
<TouchableOpacity
key={opt}
style={styles.option}
onPress={()=>update({
goal,
strictness,
sessionLength:opt,
timeSaved,
onboarded:true
})}
>
<Text style={styles.text}>
{opt}
</Text>
</TouchableOpacity>
))}
</View>

<Text style={styles.timer}>
{Math.floor(seconds/60)}:
{String(seconds%60).padStart(2,"0")}
</Text>

{locked&&(
<View style={styles.lockBox}>

<Text style={styles.lock}>
INTERVENTION
</Text>

<Text style={styles.prompt}>
{usePuzzle
?selectedPuzzle.q
:selectedPrompt}
</Text>

<TextInput
style={styles.input}
placeholder="Respond..."
placeholderTextColor="#666"
value={answer}
onChangeText={setAnswer}
/>

<TouchableOpacity
style={styles.button}
onPress={verify}
>
<Text style={styles.text}>
Submit
</Text>
</TouchableOpacity>

{solved&&(
<TouchableOpacity
style={styles.unlock}
onPress={reset}
>
<Text style={styles.text}>
Resume Intentionally
</Text>
</TouchableOpacity>
)}

</View>
)}

</View>
);
}

const styles=StyleSheet.create({
container:{
flex:1,
backgroundColor:"#050505",
justifyContent:"center",
alignItems:"center"
},
title:{
color:"#4DA6FF",
fontSize:22
},
row:{
flexDirection:"row",
flexWrap:"wrap",
justifyContent:"center",
marginVertical:20
},
option:{
backgroundColor:"#111",
padding:12,
margin:6,
borderRadius:12
},
timer:{
fontSize:52,
color:"#fff"
},
lockBox:{
marginTop:30,
width:"85%"
},
lock:{
fontSize:28,
color:"#4DA6FF",
textAlign:"center"
},
prompt:{
color:"#fff",
marginVertical:20,
fontSize:18,
textAlign:"center"
},
input:{
backgroundColor:"#111",
padding:16,
borderRadius:14,
color:"#fff"
},
button:{
backgroundColor:"#4DA6FF",
padding:16,
marginTop:20,
borderRadius:14
},
unlock:{
backgroundColor:"#222",
padding:16,
marginTop:20,
borderRadius:14
},
text:{
color:"#fff",
textAlign:"center"
}
});
