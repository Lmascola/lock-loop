import React,{
useState,
useContext,
useEffect
} from "react";

import {
View,
Text,
TouchableOpacity,
StyleSheet,
TextInput
} from "react-native";

import {AppContext} from "../context/AppContext";

export default function FocusScreen(){

const{
goal,
strictness,
sessionLength,
timeSaved,
enabled,
memory,
update
}=useContext(AppContext);

const[seconds,setSeconds]=useState(0);
const[locked,setLocked]=useState(false);
const[answer,setAnswer]=useState("");
const[solved,setSolved]=useState(false);

const prompts={
Focus:"What task must you finish?",
"Sleep Better":"What time will you sleep?",
Study:"What are you studying?",
"Reduce Addiction":"What are you avoiding?"
};

const previous=
memory.length>0
?memory[memory.length-1]
:null;

useEffect(()=>{

if(!enabled)return;

const timer=setInterval(()=>{

setSeconds(prev=>{

const next=prev+1;

if(next>=sessionLength*60&&!locked){

setLocked(true);

}

return next;

});

},1000);

return()=>clearInterval(timer);

},[enabled]);

const verify=()=>{

if(answer.length>4){

setSolved(true);

const updatedMemory=[
...memory,
{
goal,
response:answer,
time:new Date().toISOString()
}
];

update({
goal,
strictness,
sessionLength,
timeSaved:timeSaved+sessionLength,
onboarded:true,
enabled,
memory:updatedMemory
});

}

};

const resume=()=>{

setLocked(false);
setSolved(false);
setAnswer("");
setSeconds(0);

};

return(

<View style={styles.container}>

<Text style={styles.title}>
{goal} • {sessionLength}m
</Text>

<Text style={styles.timer}>
{Math.floor(seconds/60)}:
{String(seconds%60).padStart(2,"0")}
</Text>

{previous&&(
<View style={styles.memoryBox}>
<Text style={styles.memoryTitle}>
Coach Memory
</Text>

<Text style={styles.memoryText}>
Last session:
"{previous.response}"
</Text>
</View>
)}

{locked&&(

<View style={styles.lockBox}>

<Text style={styles.lock}>
INTERVENTION
</Text>

<Text style={styles.prompt}>
{prompts[goal]}
</Text>

<TextInput
style={styles.input}
placeholder="Respond..."
placeholderTextColor="#666"
value={answer}
onChangeText={setAnswer}
/>

<TouchableOpacity
style={styles.button}
onPress={verify}
>
<Text style={styles.text}>
Submit
</Text>
</TouchableOpacity>

{solved&&(
<TouchableOpacity
style={styles.unlock}
onPress={resume}
>
<Text style={styles.text}>
Resume Intentionally
</Text>
</TouchableOpacity>
)}

</View>

)}

</View>
);

}

const styles=StyleSheet.create({
container:{
flex:1,
backgroundColor:"#050505",
justifyContent:"center",
alignItems:"center",
padding:20
},
title:{
color:"#4DA6FF",
fontSize:24
},
timer:{
color:"#fff",
fontSize:52,
marginVertical:20
},
memoryBox:{
backgroundColor:"#111",
padding:18,
borderRadius:16,
width:"90%",
marginBottom:20
},
memoryTitle:{
color:"#4DA6FF",
fontWeight:"700"
},
memoryText:{
color:"#fff",
marginTop:8
},
lockBox:{
width:"90%"
},
lock:{
fontSize:28,
color:"#4DA6FF",
textAlign:"center"
},
prompt:{
color:"#fff",
fontSize:18,
marginVertical:20,
textAlign:"center"
},
input:{
backgroundColor:"#111",
padding:16,
borderRadius:14,
color:"#fff"
},
button:{
backgroundColor:"#4DA6FF",
padding:16,
marginTop:20,
borderRadius:14
},
unlock:{
backgroundColor:"#222",
padding:16,
marginTop:20,
borderRadius:14
},
text:{
color:"#fff",
textAlign:"center"
}
});
